```typescript
const API_BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:3001/api';

  },
  (error) => Promise.reject(error)
);

// Response interceptor - 에러 처리
apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);
```

## 5. 캐시 시스템 (Redis 대안)

로컬 개발에서 Redis 없이 캐싱하는 방법:

### 옵션 1: 메모리 캐시 (간단!)
```javascript
// backend/src/utils/cache.js
class MemoryCache {
  constructor() {
    this.cache = new Map();
  }

  set(key, value, ttl = 3600) {
    const expiresAt = Date.now() + (ttl * 1000);
    this.cache.set(key, { value, expiresAt });
  }

  get(key) {
    const item = this.cache.get(key);
    if (!item) return null;
    
    if (Date.now() > item.expiresAt) {
      this.cache.delete(key);
      return null;
    }
    
    return item.value;
  }

  delete(key) {
    this.cache.delete(key);
  }

  clear() {
    this.cache.clear();
  }
}

module.exports = new MemoryCache();
```

### 옵션 2: node-cache (추천!)
```bash
npm install node-cache
```

```javascript

// TTL 1시간, 체크 주기 2분
```

```javascript
// backend/src/services/queueService.js
const PQueue = require('p-queue').default;

// 동시 5개까지 처리
const ttsQueue = new PQueue({ concurrency: 5 });

async function addTTSJob(panelData) {
  return ttsQueue.add(async () => {
    console.log(`Processing TTS for panel ${panelData.id}`);
    // TTS 생성 로직
    const audioUrl = await generateTTS(panelData.script);
    return audioUrl;
  });
}

module.exports = { addTTSJob };
```

### 옵션 2: async-queue
```bash
npm install async
```

```javascript
const async = require('async');

const ttsQueue = async.queue(async (task, callback) => {
  try {
    const result = await generateTTS(task.script);
    callback(null, result);
  } catch (error) {
    callback(error);
  }
}, 5); // 동시 5개 처리

ttsQueue.push({ script: 'Hello' }, (err, result) => {
  if (err) console.error(err);
  else console.log('TTS generated:', result);
```
